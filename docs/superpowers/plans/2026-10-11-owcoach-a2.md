# OWCoach A2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Die Tabs Lernen, VOD und Stats bauen und die mit Stern markierten VOD-Findings auf der Fokus-Karte zeigen.

**Architecture:** Neue Regeln (Learnings gruppieren, Statistik pro Held/Map/Tag/Woche, Kartenstatus) kommen ins Kit und werden dort getestet. Neue SwiftData-Modelle `VODReview` und `Finding` hängen nicht über Relationships zusammen, sondern über einen String-Schlüssel (`reviewKey`), weil CloudKit Relationships nur optional erlaubt und das Löschen sonst kompliziert wird. Fokus-Aktionen (setzen, „sitzt“) liegen an einer Stelle (`FocusActions`), damit Heute und Lernen sich gleich verhalten.

**Tech Stack:** wie A1, dazu Swift Charts.

**Spec:** `docs/superpowers/specs/2026-10-11-owcoach-design.md`

## Global Constraints

- Alles aus `docs/superpowers/plans/2026-10-11-owcoach-a1.md` → Global Constraints gilt weiter
- SwiftData-Modelle: keine Relationships, jedes Attribut mit Default oder optional
- Charts: Linien 2 pt, Punkte mit Ring in Flächenfarbe, eine y-Achse pro Chart, ein Wert pro Chart; Todesursachen in fester Farbreihenfolge (Kategorie-Palette dunkel, Slots 1–6), „Fair“ in Neutralgrau, „Kaum gestorben“ nicht im Chart
- Kategorie-Farben dunkel (aus dem dataviz-Skill, validiert für den dunklen Hintergrund): `#3987e5`, `#d95926`, `#199e70`, `#c98500`, `#d55181`, `#008300`; Neutral `#7a7974`
- Beide Schemes müssen bauen: `OWCoach` und `OWCoach Local`

## Review Focus

1. **Finding auf eine Karte, die es nicht mehr gibt:** Learnings zeigt sie mit der rohen ID statt sie zu verstecken oder abzustürzen. Test: `LearningsTests.groupsKeepUnknownCards` (Task 1).
2. **Review löschen:** Seine Findings verschwinden mit, auch von der Fokus-Karte. Manuell in Task 5.

5. **Karte gleichzeitig aktiv und „sitzt“** (alte Karte wieder als Fokus gewählt): Status ist „aktiv“. Test: `CardStatusTests.activeWinsOverMastered` (Task 2).

```swift
import Foundation

public struct FindingRecord: Hashable, Sendable, Identifiable {
    public var key: String
    public var reviewKey: String
    public var cardID: String
    public var text: String
    public var moment: String?
    public var starred: Bool
    public var date: Date

    public var id: String { key }

    public init(key: String, reviewKey: String, cardID: String, text: String, moment: String?, starred: Bool, date: Date) {
        self.key = key
        self.reviewKey = reviewKey
        self.cardID = cardID
        self.text = text
        self.moment = moment
        self.starred = starred
        self.date = date
    }
}

public struct CardLearnings: Hashable, Sendable, Identifiable {
    public let cardID: String
    public let findings: [FindingRecord]

    public var id: String { cardID }
}

public enum Learnings {
    public static func forCard(_ cardID: String, in findings: [FindingRecord]) -> [FindingRecord] {
        findings.filter { $0.cardID == cardID }.sorted { ($0.starred ? 1 : 0, $0.date) > ($1.starred ? 1 : 0, $1.date) }
    }

    public static func starred(for cardID: String, in findings: [FindingRecord]) -> [FindingRecord] {
        forCard(cardID, in: findings).filter(\.starred)
    }

    public static func grouped(_ findings: [FindingRecord]) -> [CardLearnings] {
        Dictionary(grouping: findings, by: \.cardID)
            .map { CardLearnings(cardID: $0.key, findings: forCard($0.key, in: $0.value)) }
            .sorted { ($0.findings.count, $1.cardID) > ($1.findings.count, $0.cardID) }
    }
}
```

```swift
import Foundation

public enum StatsRange: String, CaseIterable, Sendable, Identifiable {
    case week, month, all

    public var id: String { rawValue }

    public var label: String {
        switch self {
        case .week: "7 Tage"
        case .month: "30 Tage"
        case .all: "Alles"
        }
    }

    public func filter(_ games: [GameRecord], now: Date, calendar: Calendar = .current) -> [GameRecord] {
        switch self {
        case .week: Stats.recent(games, days: 7, now: now, calendar: calendar)
        case .month: Stats.recent(games, days: 30, now: now, calendar: calendar)
        case .all: games
        }
    }
}

public struct GroupStat: Hashable, Sendable, Identifiable {
    public let id: String
    public let games: Int
    public let winRate: Double
    public let focusAverage: Double
}

public struct DayValue: Hashable, Sendable {
    public let day: Date
    public let value: Double
}

public struct WeekCause: Hashable, Sendable {
    public let weekStart: Date
    public let cause: DeathCause
    public let count: Int
}

extension Stats {
    public static func byHero(_ games: [GameRecord]) -> [GroupStat] {
        group(games, by: \.heroID)
    }

    public static func byMap(_ games: [GameRecord]) -> [GroupStat] {
        group(games, by: \.mapID)
    }

    public static func dailyFocus(_ games: [GameRecord], calendar: Calendar = .current) -> [DayValue] {
        playDays(games, calendar: calendar).compactMap { day in
            focusAverage(day.games).map { DayValue(day: day.day, value: $0) }
        }
    }

    public static func dailyWinRate(_ games: [GameRecord], calendar: Calendar = .current) -> [DayValue] {
        playDays(games, calendar: calendar).compactMap { day in
            winRate(day.games).map { DayValue(day: day.day, value: $0) }
        }
    }

    public static func weeklyCauses(_ games: [GameRecord], calendar: Calendar = .current) -> [WeekCause] {
        let counted = games.filter { $0.deathCause != .barelyDied }
        let byWeek = Dictionary(grouping: counted) { calendar.dateInterval(of: .weekOfYear, for: $0.date)?.start ?? calendar.startOfDay(for: $0.date) }
        return byWeek.keys.sorted().flatMap { week in
            DeathCause.allCases.compactMap { cause in
                let count = byWeek[week]!.filter { $0.deathCause == cause }.count
                return count > 0 ? WeekCause(weekStart: week, cause: cause, count: count) : nil
            }
        }
    }

    public static func firstUse(of cardID: String, in games: [GameRecord]) -> Date? {
        games.filter { $0.focusCardID == cardID }.map(\.date).min()
    }

    static func group(_ games: [GameRecord], by key: KeyPath<GameRecord, String>) -> [GroupStat] {
        Dictionary(grouping: games) { $0[keyPath: key] }
            .map { GroupStat(id: $0.key, games: $0.value.count, winRate: winRate($0.value) ?? 0, focusAverage: focusAverage($0.value) ?? 0) }
            .sorted { ($0.games, $1.id) > ($1.games, $0.id) }
    }
}
```

import Foundation
import SwiftData

enum FocusActions {
    @MainActor
    static func setFocus(_ cardID: String?, in context: ModelContext) {
        FocusPlan.current(in: context).activeCardID = cardID
        try? context.save()
    }

    @MainActor
    static func markMastered(_ cardID: String, in context: ModelContext) {
        let descriptor = FetchDescriptor<CardProgress>(predicate: #Predicate { $0.cardID == cardID })
        if let existing = try? context.fetch(descriptor), !existing.isEmpty {
            existing.forEach { $0.masteredAt = .now }
        } else {
            context.insert(CardProgress(cardID: cardID, masteredAt: .now))
        }
        let plan = FocusPlan.current(in: context)
        if plan.activeCardID == cardID { plan.activeCardID = nil }
        try? context.save()
    }

    @MainActor
    static func deleteReview(_ review: VODReview, in context: ModelContext) {
        let reviewKey = review.key
        let descriptor = FetchDescriptor<Finding>(predicate: #Predicate { $0.reviewKey == reviewKey })
        (try? context.fetch(descriptor))?.forEach(context.delete)
        context.delete(review)
        try? context.save()
    }
}

struct VODListView: View {
    @Environment(\.content) private var content
    @Environment(\.modelContext) private var context
    @Query(sort: \VODReview.date, order: .reverse) private var reviews: [VODReview]
    @Query private var findings: [Finding]
    @State private var showLearnings = false
    @State private var path: [VODReviewRoute] = []

    var body: some View {
        NavigationStack(path: $path) {
            Group {
                if showLearnings {
                    LearningsView(findings: findings.map(\.record))
                } else if reviews.isEmpty {
                    ContentUnavailableView("Noch keine Reviews", systemImage: "play.rectangle", description: Text("Tippe auf +, um ein Replay zu besprechen."))
                } else {
                    List {
                        ForEach(reviews) { review in
                            NavigationLink(value: VODReviewRoute(key: review.key)) {
                                row(review)
                            }
                        }
                        .onDelete { offsets in
                            offsets.map { reviews[$0] }.forEach { FocusActions.deleteReview($0, in: context) }
                        }
                    }
                }
            }
            .safeAreaInset(edge: .top) {
                Picker("Ansicht", selection: $showLearnings) {
                    Text("Reviews").tag(false)
                    Text("Learnings").tag(true)
                }
                .pickerStyle(.segmented)
                .padding(.horizontal, 16)
                .padding(.bottom, 8)
            }
            .navigationTitle("VOD")
            .toolbar {
                ToolbarItem(placement: .topBarTrailing) {
                    Button("Neues Review", systemImage: "plus") { addReview() }
                }
            }
            .navigationDestination(for: VODReviewRoute.self) { route in
                VODDetailView(reviewKey: route.key)
            }
            .navigationDestination(for: CardRoute.self) { CardDetailView(cardID: $0.id) }
        }
    }

    private func row(_ review: VODReview) -> some View {
        let count = findings.filter { $0.reviewKey == review.key }.count
        return HStack(spacing: 12) {
            HeroIcon(heroID: review.heroID, size: 40)
            VStack(alignment: .leading, spacing: 2) {
                Text(content.map(review.mapID)?.name ?? (review.mapID.isEmpty ? "Ohne Map" : review.mapID)).font(.headline)
                Text([review.date.formatted(date: .abbreviated, time: .omitted), review.replayCode.isEmpty ? nil : review.replayCode].compactMap { $0 }.joined(separator: " · "))
                    .font(.caption)
                    .foregroundStyle(.secondary)
            }
            Spacer()
            Text("\(count)").font(.subheadline.bold()).monospacedDigit()
            Image(systemName: "text.bubble").foregroundStyle(.secondary)
        }
        .frame(minHeight: 52)
    }

        context.insert(review)
        try? context.save()
        showLearnings = false
        path.append(VODReviewRoute(key: review.key))

    private func lineChart(title: String, points: [DayValue], domain: ClosedRange<Double>, selection: Binding<Date?>, format: @escaping (Double) -> String) -> some View {
        let selected = selection.wrappedValue.flatMap { day in points.min { abs($0.day.timeIntervalSince(day)) < abs($1.day.timeIntervalSince(day)) } }
        return VStack(alignment: .leading, spacing: 10) {
            HStack {
                Text(title).font(.headline)
                Spacer()
                if let selected {
                    Text("\(selected.day.formatted(date: .abbreviated, time: .omitted)): \(format(selected.value))")
                        .font(.caption.bold())
                        .foregroundStyle(.secondary)
                }
            }
            Chart {
                ForEach(points, id: \.day) { point in
                    LineMark(x: .value("Tag", point.day, unit: .day), y: .value(title, point.value))
                        .lineStyle(StrokeStyle(lineWidth: 2, lineCap: .round, lineJoin: .round))
                        .foregroundStyle(Theme.accent)
                    PointMark(x: .value("Tag", point.day, unit: .day), y: .value(title, point.value))
                        .symbolSize(60)
                        .foregroundStyle(Theme.accent)
                        .symbol { Circle().fill(Theme.accent).frame(width: 8, height: 8).overlay(Circle().stroke(Color(.systemBackground), lineWidth: 2)) }
                }
                if let selected {
                    RuleMark(x: .value("Tag", selected.day, unit: .day))
                        .foregroundStyle(.secondary.opacity(0.4))
                        .lineStyle(StrokeStyle(lineWidth: 1))
                }
            }
            .chartYScale(domain: domain)
            .chartXSelection(value: selection)
            .frame(height: 160)
        }
        .glassPanel()
    }

    private func causesChart(_ games: [GameRecord]) -> some View {
        let weeks = Stats.weeklyCauses(games)
        return VStack(alignment: .leading, spacing: 10) {
            Text("Todesursachen pro Woche").font(.headline)
            Chart(weeks, id: \.self) { item in
                BarMark(x: .value("Woche", item.weekStart, unit: .weekOfYear), y: .value("Games", item.count))
                    .foregroundStyle(by: .value("Ursache", item.cause.label))
                    .cornerRadius(4)
            }
            .chartForegroundStyleScale(domain: CauseColors.ordered.map(\.label), range: CauseColors.ordered.map(CauseColors.color))
            .chartLegend(position: .bottom, alignment: .leading)
            .frame(height: 200)
        }
        .glassPanel()
    }

    private func groupList<Label: View>(title: String, stats: [GroupStat], @ViewBuilder label: @escaping (String) -> Label) -> some View {

                    Text("F \(stat.focusAverage.formatted(.number.precision(.fractionLength(1))))").font(.caption).monospacedDigit().frame(width: 40, alignment: .trailing)
                }
                .frame(minHeight: 44)
            }
        }
        .glassPanel()
    }
